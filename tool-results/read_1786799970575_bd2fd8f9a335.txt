    14→// Dashboard Stats
    15→app.get('/api/dashboard/stats', async (c) => {
    16→  try {
    17→    const [hospitals, doctors, appointments, completedAppointments] = await Promise.all([
    18→      prisma.hospital.count({ where: { active: true } }),
    19→      prisma.doctor.count({ where: { available: true } }),
    20→      prisma.appointment.count({ where: { date: new Date().toISOString().split('T')[0] } }),
    21→      prisma.appointment.findMany({ where: { status: 'completed' } }),
    22→    ])
    23→    const revenue = completedAppointments.reduce((sum, a) => {
    24→      return sum + (a.doctorId ? 0 : 0)
    25→    }, 0)
    26→    const totalRevenue = await prisma.doctor.aggregate({
    27→      _sum: { price: true },
    28→    })
    29→    const completedCount = await prisma.appointment.count({ where: { status: 'completed' } })
    30→    return c.json({
    31→      totalHospitals: hospitals,
    32→      totalDoctors: doctors,
    33→      todayAppointments: appointments,
    34→      totalRevenue: (totalRevenue._sum.price || 0) * completedCount || 0,
    35→      totalDepartments: await prisma.department.count({ where: { active: true } }),
    36→      totalAppointments: await prisma.appointment.count(),
    37→    })
    38→  } catch (e: any) {
    39→    return c.json({ error: e.message }, 500)
    40→  }
    41→})

    84→    for (const doc of doctors) {
    85→      await prisma.schedule.deleteMany({ where: { doctorId: doc.id } })
    86→      await prisma.appointment.deleteMany({ where: { doctorId: doc.id } })
    87→    }
    88→    await prisma.doctor.deleteMany({ where: { hospitalId: id } })
    89→    await prisma.department.deleteMany({ where: { hospitalId: id } })
    90→    await prisma.hospital.delete({ where: { id } })
    91→    return c.json({ success: true })
    92→  } catch (e: any) {
